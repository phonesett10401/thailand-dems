# Demo Page + Video Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A `/demo` page whose top is a 35 s animated Remotion video (16:9 and 9:16, English, music), followed by plain-words copy, three role cards and an "Under the hood" section for teachers.

**Architecture:** A new sibling project, `demo-video/`, holds the Remotion source. One `Demo` component is drawn `wide` or `tall`. `npm run render` writes MP4s and poster JPGs into `web/static/demo/`. The SvelteKit page plays them with a plain `<video>`. `<source media>` lets the browser pick the 9:16 file on upright phones and 16:9 everywhere else. No Remotion code ships to visitors.

**Tech Stack:**
- Video: Remotion 4.0.534, React 19, @remotion/fonts, Fontsource (Geist, Anuphan), world-atlas + topojson-client + d3-geo (one-off map generation), Vitest 4.
- Web: SvelteKit 2 / Svelte 5 runes, Tailwind v4, Paraglide 2, Playwright + axe.

**Spec:** `docs/superpowers/specs/2026-10-11-demo-page-design.md`

## Global Constraints

### Project rules
- Commits carry **no** Co-Authored-By or other attribution lines. Never push. Never read or print any `.env`.
- Do not run `npm install` inside `web/`: Windows prunes `@emnapi` entries from `web/package-lock.json`. `demo-video/` has its own `package.json`, and installing there is expected.

### Video constants
- FPS 30, TOTAL 1050 frames (35 s).
- Wide: 1920×1080. Tall: 1080×1920.
- BPM 73.35. Beat k sits at video frame `round(k * 60 * 30 / 73.35)`.
- Music `trimBefore` = 214 frames (7.13 s into the track).
- These were measured from the track with ffmpeg on 2026-10-11:
  - lift at track 21.8 s → video frame 442
  - ease-off starts at track 39.0 s → video frame 957
  - loud again at track 41.25 s → video frame 1024. **Audio must be silent by frame 1002.**

## Review Focus

1. **Upright phone vs everything else.** An upright phone (412×915) must get the 9:16 frame and source. A landscape phone (915×412) and a tablet (1024×768) must get 16:9. Pinned in Task 7's e2e.
2. **Reduced motion.** With "reduce motion" on, the video must not start playing by itself. Pinned in Task 7's e2e via a stubbed `play()`.
3. **Music re-entering at the end.** The track gets loud again at video frame 1024. The audio fade must reach 0 by frame 1002. Pinned in Task 1 (`AUDIO.fadeOutEnd <= 1002`).
4. **Thai page.** `/th/demo` must render Thai copy and still play the English video, with no missing-key fallbacks. Pinned in Task 7 (Thai heading assertion + key-parity test).
5. **Video files missing or unplayable** (e.g. Playwright's Chromium has no H.264). The page text and buttons must still work and stay accessible. Pinned by Task 7's axe test running on Chromium, which cannot play the MP4.

`demo-video/src/timeline.ts`:
```ts
// Measured from the track (ffmpeg, 2026-10-11): 73.35 BPM; trimming 214 frames (7.13 s)
// puts a beat on video frame 0, the lift on frame 442, the ease-off on 957, and the
// track gets loud again on frame 1024.
export const FPS = 30;
export const TOTAL = 35 * FPS;
export const BPM = 73.35;
export const MUSIC_TRIM = 214;
export const beatFrame = (k: number) => Math.round((k * 60 * FPS) / BPM);

export type SceneId = 'logo' | 'map' | 'level' | 'route' | 'prepare' | 'how' | 'cta';
const BEATS: [SceneId, number][] = [['logo', 0], ['map', 4], ['level', 10], ['route', 21], ['prepare', 27], ['how', 32], ['cta', 38]];

export const SCENES = BEATS.map(([id, beat], i) => {
	const from = beatFrame(beat);
	const end = i + 1 < BEATS.length ? beatFrame(BEATS[i + 1][1]) : TOTAL;
	return { id, beat, from, duration: end - from };
});
export const scene = (id: SceneId) => SCENES.find((s) => s.id === id)!;

/** Absolute frames. Safe → Watch (haze) → Warning (heavy rain) on the music lift. */
export const LEVEL_CUES = { watch: beatFrame(13), warning: beatFrame(18) };
/** Absolute frames: go-bag items tick on beats; the screen flips to Thai on beat 30. */
export const PREP_CUES = { ticks: [beatFrame(28), beatFrame(29), beatFrame(30), beatFrame(31)], flip: beatFrame(30) };
export const AUDIO = { fadeInEnd: 10, fadeOutStart: 957, fadeOutEnd: 1000 };
```

```ts
import en from '../../web/messages/en.json';
import th from '../../web/messages/th.json';

export type Lang = 'en' | 'th';
const ALL: Record<Lang, Record<string, string>> = { en, th };

/** The app's own strings, so the phone screens in the video match the real app. */
export function msg(lang: Lang, key: string, p: Record<string, string | number> = {}) {
	const s = ALL[lang][key];
	if (s === undefined) throw new Error(`missing message ${lang}:${key}`);
	return s.replace(/\{(\w+)\}/g, (_, k: string) => String(p[k] ?? `{${k}}`));
}
```

```js
// One-off: Natural Earth 1:50m (public domain, via world-atlas) → src/thailand.ts
import { readFileSync, writeFileSync } from 'node:fs';
import { createRequire } from 'node:module';
import { feature } from 'topojson-client';
import { geoMercator, geoPath } from 'd3-geo';

const require = createRequire(import.meta.url);
const topo = JSON.parse(readFileSync(require.resolve('world-atlas/countries-50m.json'), 'utf8'));
const thailand = feature(topo, topo.objects.countries).features.find((f) => f.id === '764');
if (!thailand) throw new Error('Thailand (764) not found');

const W = 600, H = 1000;
const proj = geoMercator().fitExtent([[20, 20], [W - 20, H - 20]], thailand);
const d = geoPath(proj).digits(1)(thailand);
const at = (lon, lat) => proj([lon, lat]).map((n) => Math.round(n));
const PLACES = {
	chiangmai: at(98.98, 18.79),
	maesot: at(98.57, 16.71),
	ayutthaya: at(100.57, 14.35),
	bangkok: at(100.5, 13.76),
	nakhonsi: at(99.96, 8.43)
};
writeFileSync(
	new URL('../src/thailand.ts', import.meta.url),
	`// Generated by scripts/make-thailand.mjs from Natural Earth 1:50m (public domain). Do not edit.\n` +
		`export const THAI_VIEWBOX = '0 0 ${W} ${H}';\n` +
		`export const THAI_PATH = ${JSON.stringify(d)};\n` +
		`export const PLACES: Record<'chiangmai' | 'maesot' | 'ayutthaya' | 'bangkok' | 'nakhonsi', [number, number]> = ${JSON.stringify(PLACES)};\n`
);
console.log('wrote src/thailand.ts', d.length, 'chars');
```

- [ ] **Step 4: Shared UI.** `demo-video/src/ui.tsx`:

```tsx
import type { CSSProperties, ReactNode } from 'react';
import { AbsoluteFill, spring, useCurrentFrame, useVideoConfig } from 'remotion';
import { ARC, BODY, C, FONT, SHIELD } from './brand';
import type { Caption } from './captions';
import type { Layout } from './compositions';
import { msg, type Lang } from './i18n';

export type SceneProps = { layout: Layout; lang: Lang };

/** Fade + rise in, starting `delay` frames into the current Sequence. */
export function useAppear(delay = 0): CSSProperties {
	const f = useCurrentFrame();
	const { fps } = useVideoConfig();
	const p = spring({ frame: f - delay, fps, config: { damping: 200 } });
	return { opacity: p, transform: `translateY(${(1 - p) * 24}px)` };
}
```

```tsx
import { AbsoluteFill, interpolate, interpolateColors, spring, useCurrentFrame, useVideoConfig } from 'remotion';
import { C } from '../brand';
import { CAPTIONS, LABELS } from '../captions';
import { msg } from '../i18n';
import { LEVEL_CUES, scene } from '../timeline';
import { Phone, PhoneHeader, Stage, TabBar, type SceneProps } from '../ui';

export function LevelScene({ layout, lang }: SceneProps) {
	const f = useCurrentFrame();
	const { fps } = useVideoConfig();
	const start = scene('level').from;
	const watchAt = LEVEL_CUES.watch - start;
	const warnAt = LEVEL_CUES.warning - start;
	const state = f < watchAt ? 'safe' : f < warnAt ? 'watch' : 'warning';
	const col = interpolateColors(f, [watchAt - 6, watchAt, warnAt - 6, warnAt], [C.safe, C.watch, C.watch, C.warning]);
	const pulse = state === 'warning' ? (Math.sin((f - warnAt) / 5) + 1) / 2 : 0;
	const flash = state === 'warning' ? 1 - spring({ frame: f - warnAt, fps, config: { damping: 30 } }) : 0;
	const swap = spring({ frame: f - (state === 'safe' ? 0 : state === 'watch' ? watchAt : warnAt), fps, config: { damping: 200 } });

	// These follow the real rules: haze PM2.5 41.2 > 37.5 is Watch; 112 mm of rain ≥ 90 is Warning.
	const title =
		state === 'safe' ? msg(lang, 'now_safe_title')
		: state === 'watch' ? msg(lang, 'now_title_1', { a: msg(lang, 'haz_smoke') })
		: msg(lang, 'now_title_2', { a: msg(lang, 'haz_rain'), b: msg(lang, 'haz_smoke') });
	const lines =
		state === 'safe' ? [msg(lang, 'now_safe_body', { place: LABELS.en.place })]
		: state === 'watch' ? [msg(lang, 'line_pm25', { value: 41.2 })]
		: [msg(lang, 'line_rain', { value: 112 }), msg(lang, 'line_pm25', { value: 41.2 })];

	return (
		<Stage layout={layout} caption={CAPTIONS.en.level}>
			<Phone layout={layout}>
				<PhoneHeader arc={col} place={LABELS.en.place} glow={pulse} />
				<div style={{ padding: '28px 22px', opacity: swap, transform: `translateY(${(1 - swap) * 16}px)` }}>
					<div style={{ display: 'inline-flex', alignItems: 'center', gap: 10, padding: '10px 18px', borderRadius: 999, background: `${col}26`, color: col, fontWeight: 700, fontSize: 22 }}>
						<span style={{ width: 12, height: 12, borderRadius: 6, background: col }} />
						{msg(lang, `lvl_${state}`)}
					</div>
					<div style={{ fontSize: 36, fontWeight: 800, lineHeight: 1.1, letterSpacing: '-0.02em', marginTop: 22 }}>{title}</div>

		</Stage>
```

```tsx
import type { ReactNode } from 'react';
import { interpolate, spring, useCurrentFrame, useVideoConfig } from 'remotion';
import { C, FONT } from '../brand';
import { CAPTIONS, LABELS } from '../captions';
import { LogoMark, Stage, type SceneProps } from '../ui';

type Box = { x: number; y: number };

export function HowScene({ layout }: SceneProps) {
	const f = useCurrentFrame();
	const { fps } = useVideoConfig();
	const wide = layout === 'wide';
	const W = 900, H = wide ? 700 : 1150;
	const L = LABELS.en;
	// Sources → DEMS → outputs, left→right (wide) or top→bottom (tall).
	const col = (i: number, n: number, along: number): Box =>
		wide ? { x: along, y: (H / (n + 1)) * (i + 1) } : { x: (W / (n + 1)) * (i + 1), y: along };
	const src = L.sources.map((_, i) => col(i, 3, wide ? 150 : 140));
	const hub: Box = wide ? { x: W / 2, y: H / 2 } : { x: W / 2, y: H / 2 };
	const out = L.outputs.map((_, i) => col(i, 3, wide ? W - 150 : H - 140));
	const appear = (d: number) => spring({ frame: f - d, fps, config: { damping: 200 } });
	const flow = (f % 30) / 30;
	const lines = [...src.map((s) => [s, hub] as const), ...out.map((o, i) => [hub, o, L.outputs[i].phase2] as const)];
	const node = (b: Box, d: number, children: ReactNode, dim = false) => (
		<div style={{ position: 'absolute', left: b.x, top: b.y, transform: `translate(-50%,-50%) scale(${0.9 + 0.1 * appear(d)})`, opacity: appear(d) * (dim ? 0.45 : 1), padding: '16px 22px', borderRadius: 20, background: C.raised, border: `1px solid ${C.line}`, fontFamily: FONT, textAlign: 'center', minWidth: wide ? 220 : 240 }}>{children}</div>
	);
	return (
		<Stage layout={layout} caption={CAPTIONS.en.how}>
			<div style={{ position: 'relative', width: W, height: H }}>
				<svg width={W} height={H} style={{ position: 'absolute', inset: 0 }}>
					{lines.map(([a, b, dim], i) => (
						<g key={i} opacity={interpolate(f, [10 + i * 3, 25 + i * 3], [0, dim ? 0.35 : 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' })}>
							<line x1={a.x} y1={a.y} x2={b.x} y2={b.y} stroke="rgba(255,255,255,0.18)" strokeWidth={3} />
							<circle cx={a.x + (b.x - a.x) * flow} cy={a.y + (b.y - a.y) * flow} r={7} fill={C.info} />
						</g>
					))}
				</svg>
				{L.sources.map((s, i) => (
					<div key={s.name}>{node(src[i], i * 4, <><div style={{ fontSize: 26, fontWeight: 700 }}>{s.name}</div><div style={{ fontSize: 18, color: C.muted, marginTop: 4 }}>{s.what}</div></>)}</div>
				))}
				<div style={{ position: 'absolute', left: hub.x, top: hub.y, transform: 'translate(-50%,-50%)', opacity: appear(12) }}>
					<LogoMark size={wide ? 170 : 190} arc={C.safe} glow={0.6} />
				</div>
				{L.outputs.map((o, i) => (
					<div key={o.name}>{node(out[i], 20 + i * 4, <><div style={{ fontSize: 26, fontWeight: 700 }}>{o.name}</div>{o.phase2 && <div style={{ fontSize: 16, color: C.muted, marginTop: 4, letterSpacing: '0.08em', textTransform: 'uppercase' }}>{L.phase2}</div>}</>, o.phase2)}</div>
				))}
			</div>
		</Stage>
	);
}
```

```tsx
import { LevelScene } from './scenes/LevelScene';
import { LogoScene } from './scenes/LogoScene';

import { PrepareScene } from './scenes/PrepareScene';
import { RouteScene } from './scenes/RouteScene';

import type { SceneProps } from './ui';

const VIEWS: Record<SceneId, FC<SceneProps>> = {
	logo: LogoScene, map: MapScene, level: LevelScene, route: RouteScene, prepare: PrepareScene, how: HowScene, cta: CtaScene
};

const volume = (f: number) =>
	interpolate(f, [0, AUDIO.fadeInEnd, AUDIO.fadeOutStart, AUDIO.fadeOutEnd], [0, 1, 1, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

export function Demo(props: SceneProps) {
	return (
		<AbsoluteFill style={{ background: C.bg }}>
			{SCENES.map((s) => {
				const View = VIEWS[s.id];
				return (
					<Sequence key={s.id} from={s.from} durationInFrames={s.duration} name={s.id}>
						<View {...props} />
					</Sequence>
				);
			})}
			<Html5Audio src={staticFile('music.mp3')} trimBefore={MUSIC_TRIM} volume={volume} />

	);
}
```
